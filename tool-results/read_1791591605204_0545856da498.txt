    50→@dataclass
    51→class ClassValue:
    52→    name: str
    53→    base: Optional["ClassValue"]
    54→    methods: Dict[str, FunctionValue]
    55→    env: "Env"
    56→
    57→    def mro_methods(self) -> Dict[str, FunctionValue]:
    58→        """Resuelve métodos con herencia (hijo sobrescribe padre)."""
    59→        chain: List[ClassValue] = []
    60→        cur: Optional[ClassValue] = self
    61→        seen = set()
    62→        while cur is not None and cur.name not in seen:
    63→            chain.append(cur)
    64→            seen.add(cur.name)
    65→            cur = cur.base
    66→        merged: Dict[str, FunctionValue] = {}
    67→        for cls in reversed(chain):
    68→            merged.update(cls.methods)
    69→        return merged
    70→
    71→    def lookup(self, name: str) -> Optional[FunctionValue]:
    72→        return self.mro_methods().get(name)
    73→
    74→    def lookup_with_owner(self, name: str):
    75→        cur: Optional[ClassValue] = self
    76→        while cur is not None:
    77→            if name in cur.methods:
    78→                return cur.methods[name], cur
    79→            cur = cur.base
    80→        return None, None

   132→@dataclass
   133→class GeneratorValue:
   134→    """Generador perezoso basado en yield dentro de fn."""
   135→    fn: FunctionValue
   136→    args: list
   137→    _interp: Any = None
   138→    _exhausted: bool = False
   139→    _gen: Any = field(default=None, repr=False)
   140→
   141→    def __iter__(self):
   142→        return self
   143→
   144→    def __next__(self):
   145→        if self._exhausted:
   146→            raise StopIteration
   147→        if self._gen is None:
   148→            self._gen = self._start()
   149→        try:
   150→            return next(self._gen)
   151→        except StopIteration:
   152→            self._exhausted = True
   153→            raise
   154→
   155→    def _start(self):
   156→        local = Env(parent=self.fn.env)
   157→        for p, a in zip(self.fn.params, self.args):
   158→            local.define(p, a)
   159→        for stmt in self.fn.body:
   160→            try:
   161→                self._interp.exec_stmt(stmt, local)
   162→            except YieldSignal as ys:
   163→                yield ys.value
   164→            except ReturnSignal as rs:
   165→                if rs.value is not None:
   166→                    yield rs.value
   167→                return
   168→        return

   180→@dataclass
   181→class RecordType:
   182→    name: str
   183→    fields: list
   184→
   185→    def __call__(self, *args, **kwargs):
   186→        data = {}
   187→        if len(args) == 1 and isinstance(args[0], dict):
   188→            data = dict(args[0])
   189→        else:
   190→            for i, f in enumerate(self.fields):
   191→                if i < len(args):
   192→                    data[f] = args[i]
   193→        for k, v in kwargs.items():
   194→            data[k] = v
   195→        return RecordInstance(self.name, data)
   196→
   197→    def __repr__(self) -> str:
   198→        return f"<record {self.name}>"

   210→@dataclass
   211→class ChannelValue:
   212→    q: Any
   213→    name: str = "chan"
   214→
   215→    def send(self, value):
   216→        self.q.put(value)
   217→        return None
   218→
   219→    def recv(self, timeout=None):
   220→        import queue
   221→        try:
   222→            if timeout is None:
   223→                return self.q.get()
   224→            return self.q.get(timeout=float(timeout))
   225→        except queue.Empty:
   226→            return None
   227→
   228→    def __repr__(self) -> str:
   229→        return f"<channel {self.name}>"

   232→@dataclass
   233→class Env:
   234→    values: Dict[str, Any] = field(default_factory=dict)
   235→    parent: Optional["Env"] = None
   236→    constants: set = field(default_factory=set)
   237→
   238→    def get(self, name: str) -> Any:
   239→        if name in self.values:
   240→            return self.values[name]
   241→        if self.parent:
   242→            return self.parent.get(name)
   243→        raise AILangRuntimeError(f"nombre no definido: {name}")
   244→
   245→    def set(self, name: str, value: Any) -> None:
   246→        env: Optional[Env] = self
   247→        while env is not None:
   248→            if name in env.values:
   249→                if name in env.constants:
   250→                    raise AILangRuntimeError(f"no se puede reasignar const '{name}'")
   251→                env.values[name] = value
   252→                return
   253→            env = env.parent
   254→        raise AILangRuntimeError(f"asignación a variable no declarada: {name}")
   255→
   256→    def define(self, name: str, value: Any, *, const: bool = False) -> None:
   257→        self.values[name] = value
   258→        if const:
   259→            self.constants.add(name)

   262→def _truthy(v: Any) -> bool:
   263→    """Semántica normativa: null/false/0/0.0/""/[]/{} son falsy."""
   264→    if v is None or v is False:
   265→        return False
   266→    if isinstance(v, (int, float)) and not isinstance(v, bool) and v == 0:
   267→        return False
   268→    if isinstance(v, str) and v == "":
   269→        return False
   270→    if isinstance(v, (list, dict)) and len(v) == 0:
   271→        return False
   272→    return True

   275→class Interpreter:
   276→    def __init__(self, brain_engine: Any = None, module_path: Optional[list] = None):
   277→        self.global_env = Env()
   278→        self.budget_ms: Optional[int] = None
   279→        self.brain = brain_engine
   280→        self.module_path = list(module_path or [".", "stdlib"])
   281→        self._module_cache: Dict[str, Any] = {}
   282→        self._builtins_mods = builtin_modules()
   283→        self._install_builtins(self.global_env)
   284→        if brain_engine is not None:
   285→            brain_mod.bind_engine(brain_engine)
   286→        # Gramática unificada: módulo brain siempre en el entorno global
   287→        try:
   288→            if "brain" in self._builtins_mods:
   289→                self.global_env.define("brain", self._builtins_mods["brain"])
   290→        except Exception:
   291→            pass

   293→    def _install_builtins(self, env: Env) -> None:
   294→        def _print(*args):
   295→            print(*args)
   296→            return None
   297→
   298→        def _len(x):
   299→            return len(x)
   300→
   301→        def _str(x):
   302→            return str(x)
   303→
   304→        def _int(x):
   305→            return int(x)
   306→
   307→        def _float(x):
   308→            return float(x)
   309→
   310→        def _range(*a):
   311→            return list(range(*a))
   312→
   313→        env.define("print", _print)
   314→        env.define("len", _len)
   315→        env.define("str", _str)
   316→        env.define("int", _int)
   317→        env.define("float", _float)
   318→        env.define("range", _range)
   319→        env.define("true", True)
   320→        env.define("false", False)
   321→        env.define("null", None)
   322→
   323→        def _type(x):
   324→            if x is None:
   325→                return "null"
   326→            if isinstance(x, bool):
   327→                return "bool"
   328→            if isinstance(x, int) and not isinstance(x, bool):
   329→                return "int"
   330→            if isinstance(x, float):
   331→                return "float"
   332→            if isinstance(x, str):
   333→                return "string"
   334→            if isinstance(x, list):
   335→                return "list"
   336→            if isinstance(x, dict):
   337→                return "dict"
   338→            if isinstance(x, FunctionValue):
   339→                return "function"
   340→            if isinstance(x, ClassValue):
   341→                return "class"
   342→            if isinstance(x, Instance):
   343→                return x.cls.name
   344→            if isinstance(x, EnumMember):
   345→                return x.enum_name
   346→            if isinstance(x, RecordInstance):
   347→                return x.record_name
   348→            if isinstance(x, GeneratorValue):
   349→                return "generator"
   350→            if isinstance(x, ChannelValue):
   351→                return "channel"
   352→            if isinstance(x, Module):
   353→                return "module"
   354→            return type(x).__name__
   355→
   356→        def _isinstance(x, name):
   357→            return _type(x) == str(name)
   358→
   359→        env.define("type", _type)
   360→        env.define("isinstance", _isinstance)
   361→
   362→        def _channel(name="chan", maxsize=0):
   363→            import queue
   364→            return ChannelValue(queue.Queue(maxsize=int(maxsize) if maxsize else 0), str(name))
   365→
   366→        def _implements(obj, iface):
   367→            if not isinstance(iface, InterfaceType):
   368→                return False
   369→            if not isinstance(obj, Instance):
   370→                return False
   371→            for m in iface.methods:
   372→                meth, _ = obj.cls.lookup_with_owner(m)
   373→                if meth is None:
   374→                    return False
   375→            return True
   376→
   377→        env.define("implements", _implements)
   378→        env.define("channel", _channel)
   379→
   380→        if self.brain is not None:
   381→            eng = self.brain
   382→
   383→            def brain_ingest(source, path, content, **kw):
   384→                return eng.ingest(str(source), str(path), str(content), **kw)
   385→
   386→            def brain_query(text, top_k=10):
   387→                return eng.search(str(text), top_k=int(top_k))
   388→
   389→            def brain_think(text, top_k=8):
   390→                return eng.think(str(text), top_k=int(top_k))
   391→
   392→            def brain_graph(entity, depth=2):
   393→                return eng.graph(str(entity), depth=int(depth))
   394→
   395→            env.define("brain_ingest", brain_ingest)
   396→            env.define("brain_query", brain_query)
   397→            env.define("brain_think", brain_think)
   398→            env.define("brain_graph", brain_graph)

   400→    def load_module(self, name: str) -> Any:
   401→        if name in self._module_cache:
   402→            return self._module_cache[name]
   403→        if name in self._builtins_mods:
   404→            mod = self._builtins_mods[name]
   405→            self._module_cache[name] = mod
   406→            return mod
   407→        # archivo .ailang en module_path
   408→        rel = name.replace(".", "/") + ".ailang"
   409→        for root in self.module_path:
   410→            candidate = _Path(root) / rel
   411→            if candidate.is_file():
   412→                text = candidate.read_text(encoding="utf-8")
   413→                # Ejecutar en entorno fresco; exportar bindings globales del módulo
   414→                mod_env = Env()
   415→                self._install_builtins(mod_env)
   416→                program = parse(text)
   417→                for stmt in program.body:
   418→                    self.exec_stmt(stmt, mod_env)
   419→                exports = dict(mod_env.values)
   420→                # no reexportar builtins puros si se desea — se dejan por simplicidad
   421→                mod = Module(name, exports)
   422→                self._module_cache[name] = mod
   423→                return mod
   424→        raise AILangRuntimeError(f"módulo no encontrado: {name}")
   425→
   426→    def run(self, source: str) -> Any:
   427→        program = parse(source)
   428→        if program.budget_ms is not None:
   429→            self.budget_ms = program.budget_ms
   430→        result = None
   431→        for stmt in program.body:
   432→            result = self.exec_stmt(stmt, self.global_env)
   433→        return result
   434→
   435→    def exec_stmt(self, node: Any, env: Env) -> Any:
   436→        if isinstance(node, A.Let):
   437→            env.define(node.name, self.eval_expr(node.value, env))
   438→            return None
   439→        if isinstance(node, A.Const):
   440→            env.define(node.name, self.eval_expr(node.value, env), const=True)
   441→            return None
   442→        if isinstance(node, A.Match):
   443→            subject = self.eval_expr(node.subject, env)
   444→            for case in node.cases:
   445→                matched = False
   446→                if isinstance(case.pattern, A.Name) and case.pattern.id == "_":
   447→                    matched = True
   448→                elif isinstance(case.pattern, A.Literal):
   449→                    matched = subject == case.pattern.value
   450→                elif isinstance(case.pattern, A.Name):
   451→                    # bind name to subject
   452→                    local = Env(parent=env)
   453→                    local.define(case.pattern.id, subject)
   454→                    return self.exec_block(case.body, local)
   455→                if matched:
   456→                    return self.exec_block(case.body, env)
   457→            raise AILangRuntimeError(f"match sin caso para {subject!r}")
   458→        if isinstance(node, A.Assign):
   459→            val = self.eval_expr(node.value, env)
   460→            if isinstance(node.name, str):
   461→                env.set(node.name, val)
   462→            elif isinstance(node.name, A.Attr):
   463→                obj = self.eval_expr(node.name.value, env)
   464→                if isinstance(obj, Instance):
   465→                    obj.fields[node.name.attr] = val
   466→                elif isinstance(obj, dict):
   467→                    obj[node.name.attr] = val
   468→                else:
   469→                    raise AILangRuntimeError(
   470→                        f"no se puede asignar atributo en {type(obj).__name__}"
   471→                    )
   472→            else:
   473→                raise AILangRuntimeError("destino de asignación inválido")
   474→            return None
   475→        if isinstance(node, A.If):
   476→            if _truthy(self.eval_expr(node.test, env)):
   477→                return self.exec_block(node.body, env)
   478→            return self.exec_block(node.orelse, env)
   479→        if isinstance(node, A.While):
   480→            last = None
   481→            while _truthy(self.eval_expr(node.test, env)):
   482→                try:
   483→                    last = self.exec_block(node.body, env)
   484→                except BreakSignal:
   485→                    break
   486→                except ContinueSignal:
   487→                    continue
   488→            return last
   489→        if isinstance(node, A.For):
   490→            iterable = self.eval_expr(node.iter, env)
   491→            last = None
   492→            for item in iterable:
   493→                local = Env(parent=env)
   494→                local.define(node.target, item)
   495→                try:
   496→                    last = self.exec_block(node.body, local)
   497→                except BreakSignal:
   498→                    break
   499→                except ContinueSignal:
   500→                    continue
   501→            return last
   502→        if isinstance(node, A.Function):
   503→            env.define(
   504→                node.name,
   505→                FunctionValue(
   506→                    node.name, node.params, node.body, env,
   507→                    is_async=getattr(node, "is_async", False),
   508→                ),
   509→            )
   510→            return None
   511→        if isinstance(node, A.InterfaceDef):
   512→            env.define(node.name, InterfaceType(node.name, list(node.methods)))
   513→            return None
   514→        if isinstance(node, A.RecordDef):
   515→            env.define(node.name, RecordType(node.name, list(node.fields)))
   516→            return None
   517→        if isinstance(node, A.Spawn):
   518→            import threading
   519→            # spawn call or function invocation
   520→            def _run():
   521→                try:
   522→                    self.eval_expr(node.call, env)
   523→                except Exception as e:
   524→                    print(f"[spawn error] {e}")
   525→            th = threading.Thread(target=_run, daemon=True)
   526→            th.start()
   527→            return th
   528→        if isinstance(node, A.EnumDef):
   529→            variants = {}
   530→            et = EnumType(node.name, variants)
   531→            for v in node.variants:
   532→                variants[v] = EnumMember(node.name, v)
   533→            env.define(node.name, et)
   534→            return et
   535→        if isinstance(node, A.ClassDef):
   536→            base_cls = None
   537→            if node.base:
   538→                base_obj = env.get(node.base)
   539→                if not isinstance(base_obj, ClassValue):
   540→                    raise AILangRuntimeError(f"extends requiere una clase: {node.base}")
   541→                base_cls = base_obj
   542→            methods: Dict[str, FunctionValue] = {}
   543→            for m in node.methods:
   544→                methods[m.name] = FunctionValue(
   545→                    m.name, m.params, m.body, env, is_method=True
   546→                )
   547→            cls = ClassValue(node.name, base_cls, methods, env)
   548→            env.define(node.name, cls)
   549→            return cls
   550→        if isinstance(node, A.Break):
   551→            raise BreakSignal()
   552→        if isinstance(node, A.Continue):
   553→            raise ContinueSignal()
   554→        if isinstance(node, A.Yield):
   555→            val = None if node.value is None else self.eval_expr(node.value, env)
   556→            raise YieldSignal(val)
   557→        if isinstance(node, A.Return):
   558→            val = None if node.value is None else self.eval_expr(node.value, env)
   559→            raise ReturnSignal(val)
   560→        if isinstance(node, A.Import):
   561→            mod = self.load_module(node.module)
   562→            bind = node.alias or node.module.split(".")[-1]
   563→            env.define(bind, mod)
   564→            return mod
   565→        if isinstance(node, A.Try):
   566→            try:
   567→                return self.exec_block(node.body, env)
   568→            except ThrowSignal as ts:
   569→                local = Env(parent=env)
   570→                local.define(node.catch_name, ts.value)
   571→                return self.exec_block(node.catch_body, local)
   572→            except AILangRuntimeError as re:
   573→                local = Env(parent=env)
   574→                local.define(node.catch_name, str(re))
   575→                return self.exec_block(node.catch_body, local)
   576→        if isinstance(node, A.Throw):
   577→            raise ThrowSignal(self.eval_expr(node.value, env))
   578→        if isinstance(node, A.ExprStmt):
   579→            return self.eval_expr(node.value, env)
   580→        if isinstance(node, A.Block):
   581→            return self.exec_block(node.body, env)
   582→        raise AILangRuntimeError(f"statement desconocido: {type(node)}")
   583→
   584→    def exec_block(self, body: List[Any], env: Env) -> Any:
   585→        local = Env(parent=env)
   586→        last = None
   587→        for stmt in body:
   588→            last = self.exec_stmt(stmt, local)
   589→        return last

   593→    def _call_function(
   594→        self,
   595→        fn: FunctionValue,
   596→        args: list,
   597→        *,
   598→        self_obj: Any = None,
   599→        defining_cls: Any = None,
   600→    ) -> Any:
   601→        """Invoca función/método con cierre léxico, self/super y stack frames."""
   602→        expected = list(fn.params)
   603→        if len(args) != len(expected):
   604→            raise AILangRuntimeError(
   605→                f"{fn.name} espera {len(expected)} args, recibió {len(args)}"
   606→            )
   607→        local = Env(parent=fn.env)
   608→        if self_obj is not None:
   609→            local.define("self", self_obj)
   610→            if isinstance(self_obj, Instance):
   611→                owner = defining_cls or self_obj.cls
   612→                local.define("super", SuperProxy(self_obj, owner))
   613→        for p, a in zip(expected, args):
   614→            local.define(p, a)
   615→        # Si el cuerpo tiene yield → generador
   616→        def _has_yield(stmts):
   617→            for s in stmts:
   618→                if isinstance(s, A.Yield):
   619→                    return True
   620→                for attr in ("body", "orelse", "catch_body"):
   621→                    if hasattr(s, attr) and _has_yield(getattr(s, attr) or []):
   622→                        return True
   623→            return False
   624→        if _has_yield(fn.body):
   625→            return GeneratorValue(fn, list(args), _interp=self)
   626→        try:
   627→            self.exec_block(fn.body, local)
   628→        except ReturnSignal as rs:
   629→            return rs.value
   630→        except AILangRuntimeError as exc:
   631→            raise exc.push(fn.name or "<fn>") from None
   632→        return None
   633→
   634→    def _py_callable(self, value: Any) -> Any:
   635→        """Envuelve FunctionValue para que stdlib Python pueda invocarla."""
   636→        if not isinstance(value, FunctionValue):
   637→            return value
   638→        fv = value
   639→        interp = self
   640→
   641→        def _call(*args):
   642→            local = Env(parent=fv.env)
   643→            if len(args) != len(fv.params):
   644→                raise AILangRuntimeError(
   645→                    f"{fv.name} espera {len(fv.params)} args, recibió {len(args)}"
   646→                )
   647→            for p, a in zip(fv.params, args):
   648→                local.define(p, a)
   649→            try:
   650→                interp.exec_block(fv.body, local)
   651→            except ReturnSignal as rs:
   652→                return rs.value
   653→            return None
   654→
   655→        return _call
   656→
   657→    def eval_expr(self, node: Any, env: Env) -> Any:
   658→        if isinstance(node, A.InterpolatedString):
   659→            out = []
   660→            for part in node.parts:
   661→                if isinstance(part, A.Literal):
   662→                    out.append(str(part.value))
   663→                else:
   664→                    out.append(str(self.eval_expr(part, env)))
   665→            return "".join(out)
   666→        if isinstance(node, A.Await):
   667→            val = self.eval_expr(node.value, env)
   668→            # await on async function call result: if already value, return
   669→            # if FunctionValue async without call — error
   670→            return val
   671→        if isinstance(node, A.Literal):
   672→            return node.value
   673→        if isinstance(node, A.Lambda):
   674→            return FunctionValue("<lambda>", node.params, node.body, env)
   675→        if isinstance(node, A.Name):
   676→            return env.get(node.id)
   677→        if isinstance(node, A.ListLit):
   678→            return [self.eval_expr(e, env) for e in node.elts]
   679→        if isinstance(node, A.DictLit):
   680→            return {
   681→                self.eval_expr(k, env): self.eval_expr(v, env)
   682→                for k, v in zip(node.keys, node.values)
   683→            }
   684→        if isinstance(node, A.UnaryOp):
   685→            v = self.eval_expr(node.operand, env)
   686→            if node.op == "-":
   687→                return -v
   688→            if node.op == "+":
   689→                return +v
   690→            if node.op == "not":
   691→                return not _truthy(v)
   692→            raise AILangRuntimeError(f"unary {node.op}")
   693→        if isinstance(node, A.BinOp):
   694→            if node.op == "and":
   695→                left = self.eval_expr(node.left, env)
   696→                return self.eval_expr(node.right, env) if _truthy(left) else left
   697→            if node.op == "or":
   698→                left = self.eval_expr(node.left, env)
   699→                return left if _truthy(left) else self.eval_expr(node.right, env)
   700→            l = self.eval_expr(node.left, env)
   701→            r = self.eval_expr(node.right, env)
   702→            if node.op == "+":
   703→                if isinstance(l, str) or isinstance(r, str):
   704→                    return str(l) + str(r)
   705→                return l + r
   706→            ops = {
   707→                "+": lambda a, b: a + b,
   708→                "-": lambda a, b: a - b,
   709→                "*": lambda a, b: a * b,
   710→                "/": lambda a, b: a / b,
   711→                "%": lambda a, b: a % b,
   712→                "==": lambda a, b: a == b,
   713→                "!=": lambda a, b: a != b,
   714→                "<": lambda a, b: a < b,
   715→                "<=": lambda a, b: a <= b,
   716→                ">": lambda a, b: a > b,
   717→                ">=": lambda a, b: a >= b,
   718→            }
   719→            if node.op not in ops:
   720→                raise AILangRuntimeError(f"operador {node.op}")
   721→            try:
   722→                return ops[node.op](l, r)
   723→            except TypeError as e:
   724→                raise AILangRuntimeError(str(e)) from e
   725→        if isinstance(node, A.Attr):
   726→            obj = self.eval_expr(node.value, env)
   727→            if isinstance(obj, Module):
   728→                return getattr(obj, node.attr)
   729→            if isinstance(obj, RecordInstance):
   730→                if node.attr in obj.fields:
   731→                    return obj.fields[node.attr]
   732→                raise AILangRuntimeError(f"record sin campo '{node.attr}'")
   733→            if isinstance(obj, ChannelValue):
   734→                if node.attr == "send":
   735→                    return obj.send
   736→                if node.attr == "recv":
   737→                    return obj.recv
   738→            if isinstance(obj, Instance):
   739→                if node.attr in obj.fields:
   740→                    return obj.fields[node.attr]
   741→                meth, owner = obj.cls.lookup_with_owner(node.attr)
   742→                if meth is not None and owner is not None:
   743→                    return BoundMethod(obj, meth, owner)
   744→                raise AILangRuntimeError(
   745→                    f"'{obj.cls.name}' no tiene atributo '{node.attr}'"
   746→                )
   747→            if isinstance(obj, SuperProxy):
   748→                base = obj.start_cls.base
   749→                while base is not None:
   750→                    if node.attr in base.methods:
   751→                        return BoundMethod(obj.instance, base.methods[node.attr], base)
   752→                    base = base.base
   753→                raise AILangRuntimeError(f"super no tiene método '{node.attr}'")
   754→            if isinstance(obj, EnumType):
   755→                if node.attr in obj.variants:
   756→                    return obj.variants[node.attr]
   757→                raise AILangRuntimeError(f"enum {obj.name} no tiene {node.attr}")
   758→            if isinstance(obj, ClassValue):
   759→                meth = obj.lookup(node.attr)
   760→                if meth is not None:
   761→                    return meth
   762→            if isinstance(obj, dict) and node.attr in obj:
   763→                return obj[node.attr]
   764→            if hasattr(obj, node.attr) and not node.attr.startswith("_"):
   765→                return getattr(obj, node.attr)
   766→            raise AILangRuntimeError(f"atributo '{node.attr}' no encontrado en {obj!r}")
   767→        if isinstance(node, A.Call):
   768→            callee = self.eval_expr(node.func, env)
   769→            args = [self.eval_expr(a, env) for a in node.args]
   770→            # constructor: ClassName(args)
   771→            if isinstance(callee, ClassValue):
   772→                inst = Instance(callee, {})
   773→                init = callee.lookup("init")
   774→                if init is not None:
   775→                    self._call_function(init, args, self_obj=inst, defining_cls=callee)
   776→                return inst
   777→            if isinstance(callee, BoundMethod):
   778→                return self._call_function(
   779→                    callee.func, args, self_obj=callee.instance, defining_cls=callee.defining_cls
   780→                )
   781→            if isinstance(callee, FunctionValue):
   782→                return self._call_function(callee, args)
   783→            if callable(callee):
   784→                py_args = [self._py_callable(a) for a in args]
   785→                return callee(*py_args)
   786→            raise AILangRuntimeError(f"no invocable: {callee!r}")
   787→        if isinstance(node, A.Index):
   788→            base = self.eval_expr(node.value, env)
   789→            idx = self.eval_expr(node.index, env)
   790→            return base[idx]
   791→        raise AILangRuntimeError(f"expresión desconocida: {type(node)}")
   792→

   794→def run_source(
   795→    source: str,
   796→    *,
   797→    brain_engine: Any = None,
   798→    module_path: Optional[list] = None,
   799→    use_vm: bool = False,
   800→    typecheck_strict: Optional[bool] = None,
   801→) -> Any:
   802→    import os
   803→    if typecheck_strict is not None:
   804→        from ailang.lang.typecheck import typecheck
   805→        diags = typecheck(source, strict=typecheck_strict)
   806→        errors = [d for d in diags if d.level == "error"]
   807→        if errors:
   808→            raise AILangRuntimeError("typecheck: " + "; ".join(str(e) for e in errors))
   809→    # AILANG_VM=1 fuerza VM; use_vm=True también
   810→    if use_vm or os.environ.get("AILANG_VM") == "1":
   811→        from ailang.lang.vm import run_vm
   812→        return run_vm(source, module_path=module_path)
   813→    return Interpreter(brain_engine=brain_engine, module_path=module_path).run(source)
   814→
   815→
   816→def create_repl(brain_engine: Any = None) -> None:
   817→    """REPL interactivo del núcleo AILang."""
   818→    interp = Interpreter(brain_engine=brain_engine)
   819→    print(f"AILang REPL — núcleo MVP. Ctrl+D / exit para salir.")
   820→    buf: List[str] = []
   821→    while True:
   822→        try:
   823→            prompt = "... " if buf else "ailang> "
   824→            line = input(prompt)
   825→        except EOFError:
   826→            print()
   827→            break
   828→        if line.strip() in ("exit", "quit") and not buf:
   829→            break
   830→        buf.append(line)
   831→        text = "\n".join(buf)
   832→        # continue if block seems open
   833→        if text.count("{") > text.count("}"):
   834→            continue
   835→        buf = []
   836→        try:
   837→            result = interp.run(text)
   838→            if result is not None:
   839→                print(repr(result))
   840→        except (AILangSyntaxError, AILangRuntimeError) as e:
   841→            print(f"! {e}")
   842→        except Exception as e:
   843→            print(f"! {type(e).__name__}: {e}")
